\subsection{Orthonormal Bases}

\begin{exercise}{6b1}
    Suppose $e_1, \ldots, e_m$ is a list of vectors in $V$ such that
    \[
        \norm{a_1 e_1 + \cdots + a_m e_m}^2 = |a_1|^2 + \cdots + |a_m|^2
    \]
    for all $a_1, \ldots, a_m \in \fbb$.
    Show that $e_1, \ldots, e_m$ is an orthonormal list.

    \note{This exercise provides a converse to 6.24.}
\end{exercise}

\begin{sol}
    For each $k \in \set{1, \ldots, m}$, let $a_k = 1$ and $a_j = 0$ for all $j \neq k$.
    Then
    \[
        \norm{e_k}^2 = \norm{a_1 e_1 + \cdots + a_m e_m}^2 = |a_1|^2 + \cdots + |a_m|^2 = 1.
    \]
    Next, for distinct $j, k \in \set{1, \ldots, m}$, set $a_j = a_k = 1$ and $a_i = 0$ for all $i \notin \set{j, k}$.
    Then $\norm{e_j + e_k}^2 = 2$ by hypothesis, and expanding gives
    \[
        \norm{e_j + e_k}^2 = \norm{e_j}^2 + \gen{e_j, e_k} + \bar{\gen{e_j, e_k}} + \norm{e_k}^2 = 2 + 2\Re\gen{e_j, e_k},
    \]
    so $\Re\gen{e_j, e_k} = 0$.
    If $\fbb = \rbb$, then $\gen{e_j, e_k} = 0$.
    If $\fbb = \cbb$, set $a_j = 1$, $a_k = i$, and all other scalars to $0$.
    Then $\norm{e_j + ie_k}^2 = 2$ by hypothesis, and expanding gives
    \[
        \norm{e_j + ie_k}^2 = \norm{e_j}^2 + i\gen{e_k, e_j} + \bar{i\gen{e_k, e_j}} + \norm{e_k}^2 = 2 - 2\Im\gen{e_k, e_j},
    \]
    so $\Im\gen{e_j, e_k} = 0$, and thus $\gen{e_j, e_k} = 0$.
\end{sol}

\begin{exercise}{6b2}
    \begin{subexercises}
        \item Suppose $\theta \in \rbb$.
        Show that both
        \[
            (\cos\theta, \sin\theta),\, (-\sin\theta, \cos\theta) \quad \text{and} \quad (\cos\theta, \sin\theta),\, (\sin\theta, -\cos\theta)
        \]
        are orthonormal bases of $\rbb^2$.
        \item Show that each orthonormal basis of $\rbb^2$ is of the form given by one of the two possibilities in (a).
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Let $v_1 = (\cos\theta, \sin\theta)$ and $v_2 = (-\sin\theta, \cos\theta)$.
    Then $\norm{v_1}^2 = \cos^2\theta + \sin^2\theta = 1$ and $\norm{v_2}^2 = \sin^2\theta + \cos^2\theta = 1$.
    Moreover,
    \[
        \gen{v_1, v_2} = -\cos\theta\sin\theta + \sin\theta\cos\theta = 0.
    \]
    Since $v_1, v_2$ is an orthonormal list of two vectors in $\rbb^2$, it is an orthonormal basis of $\rbb^2$.
    Similarly, let $w_1 = (\cos\theta, \sin\theta)$ and $w_2 = (\sin\theta, -\cos\theta)$.
    Then $\norm{w_1}^2 = \cos^2\theta + \sin^2\theta = 1$ and $\norm{w_2}^2 = \sin^2\theta + \cos^2\theta = 1$.
    Again,
    \[
        \gen{w_1, w_2} = \cos\theta\sin\theta - \sin\theta\cos\theta = 0.
    \]
    Since $w_1, w_2$ is an orthonormal list of two vectors in $\rbb^2$, it is an orthonormal basis of $\rbb^2$.
    \item Suppose $u_1, u_2$ is an orthonormal basis of $\rbb^2$.
    If $u_1 = (x_1, y_1)$, then $\norm{u_1}^2 = x_1^2 + y_1^2 = 1$, so there exists $\theta \in \rbb$ such that $x_1 = \cos\theta$ and $y_1 = \sin\theta$.
    If $u_2 = (x_2, y_2)$, then $\norm{u_2}^2 = x_2^2 + y_2^2 = 1$ and $\gen{u_1, u_2} = x_1x_2 + y_1y_2 = 0$.
    We now solve the system of equations
    \[
        \begin{cases}
            x_2^2 + y_2^2 = 1, \\
            \cos\theta x_2 + \sin\theta y_2 = 0.
        \end{cases}
    \]
    If $\sin\theta = 0$, then $\theta = 0$ or $\theta = \pi$, so $\cos\theta = \pm 1$.
    Then $x_2 = 0$, so $y_2^2 = 1$, and thus $y_2 = \pm 1$.
    Then $u_2 = (0, \pm 1)$, which is of the form given in (a).
    
    Otherwise, $y_2 = -\frac{\cos\theta}{\sin\theta}x_2$, so
    \[
        x_2^2 + \left(-\frac{\cos\theta}{\sin\theta}x_2\right)^2 = 1 \implies x_2^2 + \frac{\cos^2\theta}{\sin^2\theta}x_2^2 = 1 \implies x_2^2\left(1 + \frac{\cos^2\theta}{\sin^2\theta}\right) = 1.
    \]
    We then have $x_2^2 = \sin^2\theta$, so $x_2 = \pm \sin\theta$.
    Then $y_2 = -\frac{\cos\theta}{\sin\theta}(\pm \sin\theta) = \mp \cos\theta$.
    Thus, $u_2 = (\pm \sin\theta, \mp \cos\theta)$, which is of the form given in (a).
\end{solenum}

\begin{exercise}{6b3}
    Suppose $e_1, \ldots, e_m$ is an orthonormal list in $V$ and $v \in V$.
    Prove that
    \[
        \norm{v}^2 = |\gen{v, e_1}|^2 + \cdots + |\gen{v, e_m}|^2 \iff v \in \spn(e_1, \ldots, e_m).
    \]
\end{exercise}

\begin{sol}
    If $v \in \spn(e_1, \ldots, e_m)$, then there exist scalars $a_1, \ldots, a_m$ such that $v = a_1 e_1 + \cdots + a_m e_m$.
    Then
    \[
        \norm{v}^2 = \norm{a_1 e_1 + \cdots + a_m e_m}^2 = |a_1|^2 + \cdots + |a_m|^2 = |\gen{v, e_1}|^2 + \cdots + |\gen{v, e_m}|^2.
    \]

    Conversely, suppose $\norm{v}^2 = |\gen{v, e_1}|^2 + \cdots + |\gen{v, e_m}|^2$.
    Let $w = v - (\gen{v, e_1}e_1 + \cdots + \gen{v, e_m}e_m)$.
    Note that $\gen{w, e_k} = \gen{v, e_k} - \gen{v, e_k} = 0$ for each $k = 1, \ldots, m$, so $w$ is orthogonal to each $e_k$.
    Then
    \[
        \norm{v}^2 = \norm*{w + \sum_{k=1}^m \gen{v, e_k}e_k}^2 = \norm{w}^2 + \sum_{k=1}^m |\gen{v, e_k}|^2.
    \]
    The assumption of $\norm{v}^2 = |\gen{v, e_1}|^2 + \cdots + |\gen{v, e_m}|^2$ gives $\norm{w}^2 = 0$, so $w = 0$, and $v \in \spn(e_1, \ldots, e_m)$.
\end{sol}

\begin{exercise}{6b4}
    Suppose $n$ is a positive integer.
    Prove that
    \[
        \frac{1}{\sqrt{2\pi}},\, \frac{\cos x}{\sqrt{\pi}},\, \frac{\cos 2x}{\sqrt{\pi}},\, \ldots,\, \frac{\cos nx}{\sqrt{\pi}},\, \frac{\sin x}{\sqrt{\pi}},\, \frac{\sin 2x}{\sqrt{\pi}},\, \ldots,\, \frac{\sin nx}{\sqrt{\pi}}
    \]
    is an orthonormal list of vectors in $C[-\pi, \pi]$, the vector space of continuous real-valued functions on $[-\pi, \pi]$ with inner product
    \[
        \gen{f, g} = \int_{-\pi}^{\pi} fg.
    \]
    \hint{The following formulas should help.}
    \begin{align*}
        (\sin x)(\cos y) &= \frac{\sin(x - y) + \sin(x + y)}{2}, \\
        (\sin x)(\sin y) &= \frac{\cos(x - y) - \cos(x + y)}{2}, \\
        (\cos x)(\cos y) &= \frac{\cos(x - y) + \cos(x + y)}{2}.
    \end{align*}
\end{exercise}

\begin{sol}
    Define the vectors
    \[
        f_0 = \frac{1}{\sqrt{2\pi}}, \quad f_k = \frac{\cos kx}{\sqrt{\pi}}, \quad g_k = \frac{\sin kx}{\sqrt{\pi}} \quad \text{for } k = 1, \ldots, n.
    \]
    Then
    \[
        \norm{f_0}^2 = \frac{1}{2\pi}\int_{-\pi}^{\pi} 1 \dd x = 1.
    \]
    Moreover, note the following for $k = 1, \ldots, n$,
    \[
        \gen{f_0, f_k} = \frac{1}{\sqrt{2\pi}}\frac{1}{\sqrt{\pi}}\int_{-\pi}^{\pi} \cos kx \dd x = \frac{1}{\sqrt{2\pi}}\frac{1}{\sqrt{\pi}}\left.\frac{\sin kx}{k}\right|_{-\pi}^{\pi} = 0,
    \]
    and $\gen{f_0, g_k} = 0$ because $\sin kx$ is an odd function.
    Hence, $f_0$ is orthogonal to each of the other vectors in the list.
    Using the $\cos kx$ and $\sin kx$ computations in the above integrals, we may simplify the calculations of the norms of $f_k$ and $g_k$ to get
    \begin{align*}
        \norm{f_k}^2 &= \frac{1}{\pi}\int_{-\pi}^{\pi} \cos^2 kx \dd x = \frac{1}{2\pi}\int_{-\pi}^{\pi} (1 + \cos 2kx) \dd x = 1, \\
        \norm{g_k}^2 &= \frac{1}{\pi}\int_{-\pi}^{\pi} \sin^2 kx \dd x = \frac{1}{2\pi}\int_{-\pi}^{\pi} (1 - \cos 2kx) \dd x = 1.
    \end{align*}
    We now show that the remaining vectors are pairwise orthogonal.
    For $j, k \in \set{1, \ldots, n}$ with $j \neq k$, we have that $j - k$ and $j + k$ are nonzero integers.
    Note that $\sin mx = 0$ for all integers $m$ when $x = \pm \pi$.
    Then
    \begin{align*}
        \gen{f_j, f_k} &= \frac{1}{\pi}\int_{-\pi}^{\pi} \cos jx \cos kx \dd x \\
        &= \frac{1}{2\pi}\int_{-\pi}^{\pi} (\cos(j - k)x + \cos(j + k)x) \dd x \\
        &= \frac{1}{2\pi}\left.\left(\frac{\sin(j - k)x}{j - k} + \frac{\sin(j + k)x}{j + k}\right)\right|_{-\pi}^{\pi} = 0, \\
        \gen{g_j, g_k} &= \frac{1}{\pi}\int_{-\pi}^{\pi} \sin jx \sin kx \dd x \\
        &= \frac{1}{2\pi}\int_{-\pi}^{\pi} (\cos(j - k)x - \cos(j + k)x) \dd x \\
        &= \frac{1}{2\pi}\left.\left(\frac{\sin(j - k)x}{j - k} - \frac{\sin(j + k)x}{j + k}\right)\right|_{-\pi}^{\pi} = 0.
    \end{align*}
    For $j, k = 1, \ldots, n$, note that $f_j g_k$ is a product of an even function and an odd function, so $\gen{f_j, g_k} = 0$.
    Thus, the list of vectors is orthonormal.
\end{sol}

\begin{exercise}{6b5}
    Suppose $f : [-\pi, \pi] \to \rbb$ is continuous.
    For each nonnegative integer $k$, define
    \[
        a_k = \frac{1}{\sqrt{\pi}} \int_{-\pi}^{\pi} f(x) \cos(kx)\dd x \quad \text{and} \quad b_k = \frac{1}{\sqrt{\pi}} \int_{-\pi}^{\pi} f(x) \sin(kx)\dd x.
    \]
    Prove that
    \[
        \frac{a_0^2}{2} + \sum_{k=1}^{\infty} (a_k^2 + b_k^2) \leq \int_{-\pi}^{\pi} f^2.
    \]
    \note{The inequality above is actually an equality for all continuous functions $f : [-\pi, \pi] \to \rbb$.
    However, proving that this inequality is an equality involves Fourier series techniques beyond the scope of this book.}
\end{exercise}

\begin{sol}
    The list of vectors in \exref{6b4} is an orthonormal list in $C[-\pi, \pi]$.
    Using the same naming conventions of $f_0, f_k, g_k$ as in \exref{6b4}, we have that
    \[
        \gen{f, f_0} = \frac{a_0}{\sqrt{2}}, \quad
        \gen{f, f_k} = a_k, \quad
        \gen{f, g_k} = b_k.
    \]
    By Bessel's inequality, for every $n \geq 1$,
    \[
        \frac{a_0^2}{2} + \sum_{k=1}^{n}(a_k^2 + b_k^2) = |\gen{f, f_0}|^2 + \sum_{k=1}^{n}(|\gen{f, f_k}|^2 + |\gen{f, g_k}|^2) \leq \norm{f}^2 = \int_{-\pi}^{\pi} f^2.
    \]
    Since the right side is independent of $n$, taking $n \to \infty$ gives the result.
\end{sol}

\begin{exercise}{6b6}
    Suppose $e_1, \ldots, e_n$ is an orthonormal basis of $V$.
    \begin{subexercises}
        \item Prove that if $v_1, \ldots, v_n$ are vectors in $V$ such that
        \[
            \norm{e_k - v_k} < \frac{1}{\sqrt{n}}
        \]
        for each $k$, then $v_1, \ldots, v_n$ is a basis of $V$.
        \item Show that there exist $v_1, \ldots, v_n \in V$ such that
        \[
            \norm{e_k - v_k} \leq \frac{1}{\sqrt{n}}
        \]
        for each $k$, but $v_1, \ldots, v_n$ is not linearly independent.
    \end{subexercises}
    \note{This exercise states in (a) that an appropriately small perturbation of an orthonormal basis is a basis.
    Then (b) shows that the number $1/\sqrt{n}$ on the right side of the inequality in (a) cannot be improved upon.}
\end{exercise}

\begin{solenum}
    \item Suppose $v_1, \ldots, v_n$ are linearly dependent.
    Then there exist scalars $a_1, \ldots, a_n$, not all zero, such that
    \[
        a_1 v_1 + \cdots + a_n v_n = 0.
    \]
    Then
    \begin{align*}
        \sqrt{\sum_{k=1}^n|a_k|^2} &= \norm*{\sum_{k=1}^n a_k e_k} \\
        &= \norm*{\sum_{k=1}^na_k(e_k - v_k)} \\
        &\leq \sum_{k=1}^n |a_k|\norm{e_k - v_k} \\
        &< \frac{1}{\sqrt{n}}\sum_{k=1}^n |a_k| \\
        &\leq \frac{1}{\sqrt{n}}\sqrt{n}\sqrt{\sum_{k=1}^n|a_k|^2} \\
        &= \sqrt{\sum_{k=1}^n|a_k|^2}.
    \end{align*}
    This results in $1 < 1$, which is a contradiction, so $v_1, \ldots, v_n$ must be linearly independent.
    Since $\dim V = n$, it follows that $v_1, \ldots, v_n$ is a basis of $V$.
    \item Consider the vector
    \[
        u = \frac{1}{\sqrt{n}}(e_1 + \cdots + e_n), \quad \norm{u}^2 = \frac{1}{n}\norm{e_1 + \cdots + e_n}^2 = \frac{1}{n}(n) = 1.
    \]
    
    For each $k = 1, \ldots, n$, define
    \[
        v_k = e_k - \frac{1}{\sqrt{n}}u \implies \norm{e_k - v_k} = \norm{\frac{1}{\sqrt{n}}u} = \frac{1}{\sqrt{n}}\norm{u} = \frac{1}{\sqrt{n}}.
    \]
    Then
    \[
        v_1 + \cdots + v_n = (e_1 - \frac{1}{\sqrt{n}}u) + \cdots + (e_n - \frac{1}{\sqrt{n}}u) = (e_1 + \cdots + e_n) - \frac{n}{\sqrt{n}}u = (e_1 + \cdots + e_n) - \sqrt{n}u = 0,
    \]
    so $v_1, \ldots, v_n$ is linearly dependent.
\end{solenum}

\begin{exercise}{6b7}
    Suppose $T \in \lin{\rbb^3}$ has an upper-triangular matrix with respect to the basis $(1, 0, 0)$, $(1, 1, 1)$, $(1, 1, 2)$.
    Find an orthonormal basis of $\rbb^3$ with respect to which $T$ has an upper-triangular matrix.
\end{exercise}

\begin{sol}
    Using the Gram--Schmidt procedure, we can produce an orthonormal basis of $\rbb^3$ from the given basis.
    Let $v_1 = (1, 0, 0), v_2 = (1, 1, 1), v_3 = (1, 1, 2)$.
    Set $f_1 = v_1 = (1, 0, 0)$, so $\norm{f_1}^2 = 1^2 + 0^2 + 0^2 = 1$.
    Then
    \[
        f_2 = v_2 - \frac{\gen{v_2, f_1}}{\norm{f_1}^2}f_1 = (1, 1, 1) - \frac{1}{1}(1, 0, 0) = (0, 1, 1)
    \]
    with $\norm{f_2}^2 = 0^2 + 1^2 + 1^2 = 2$.
    Finally,
    \[
        f_3 = v_3 - \frac{\gen{v_3, f_1}}{\norm{f_1}^2}f_1 - \frac{\gen{v_3, f_2}}{\norm{f_2}^2}f_2 = (1, 1, 2) - (1, 0, 0) - \frac{3}{2}(0, 1, 1) = \left(0, -\frac{1}{2}, \frac{1}{2}\right)
    \]
    with $\norm{f_3}^2 = 0^2 + (-\frac{1}{2})^2 + (\frac{1}{2})^2 = \frac{1}{2}$.
    Then the orthonormal basis of $\rbb^3$ is
    \[
        e_1 = \frac{f_1}{\norm{f_1}} = (1, 0, 0), \quad
        e_2 = \frac{f_2}{\norm{f_2}} = \frac{1}{\sqrt{2}}(0, 1, 1), \quad
        e_3 = \frac{f_3}{\norm{f_3}} = \sqrt{2}\left(0, -\frac{1}{2}, \frac{1}{2}\right) = \left(0, -\frac{\sqrt{2}}{2}, \frac{\sqrt{2}}{2}\right).
    \]
    By the Gram--Schmidt procedure, we know $\spn(e_1, \ldots, e_k) = \spn(v_1, \ldots, v_k)$ for each $k = 1, 2, 3$.
    Since $\mcal(T, (v_1, v_2, v_3))$ is upper-triangular, $\spn(v_1, \ldots, v_k)$ is invariant under $T$ for each $k = 1, 2, 3$.
    Hence, $\spn(e_1, \ldots, e_k)$ is invariant under $T$ for each $k = 1, 2, 3$, so $\mcal(T, (e_1, e_2, e_3))$ is upper-triangular.
\end{sol}

\begin{exercise}{6b8}
    Make $\pcal_2(\rbb)$ into an inner product space by defining $\gen{p, q} = \int_0^1 pq$ for all $p, q \in \pcal_2(\rbb)$.
    \begin{subexercises}
        \item Apply the Gram--Schmidt procedure to the basis $1, x, x^2$ to produce an orthonormal basis of $\pcal_2(\rbb)$.
        \item The differentiation operator (the operator that takes $p$ to $p'$) on $\pcal_2(\rbb)$ has an upper-triangular matrix with respect to the basis $1, x, x^2$, which is not an orthonormal basis.
        Find the matrix of the differentiation operator on $\pcal_2(\rbb)$ with respect to the orthonormal basis produced in (a) and verify that this matrix is upper triangular, as expected from the proof of 6.37.
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Let $v_1 = 1, v_2 = x, v_3 = x^2$.
    Then $f_1 = v_1 = 1$, so $\norm{f_1}^2 = \int_0^1 1^2 = 1$.
    Next,
    \[
        f_2 = v_2 - \frac{\gen{v_2, f_1}}{\norm{f_1}^2}f_1 = x - \int_0^1 x \cdot 1 = x - \frac{1}{2}
    \]
    with $\norm{f_2}^2 = \int_0^1 (x - \frac{1}{2})^2 \dd x = \frac{1}{12}$.
    Finally,
    \[
        f_3 = v_3 - \frac{\gen{v_3, f_1}}{\norm{f_1}^2}f_1 - \frac{\gen{v_3, f_2}}{\norm{f_2}^2}f_2 = x^2 - \int_0^1 x^2 \cdot 1 \dd x - 12\left(\int_0^1 x^2\left(x - \frac{1}{2}\right) \dd x\right)\left(x - \frac{1}{2}\right) = x^2 - x + \frac{1}{6}
    \]
    with $\norm{f_3}^2 = \int_0^1 (x^2 - x + \frac{1}{6})^2 \dd x = \frac{1}{180}$.
    Then the orthonormal basis of $\pcal_2(\rbb)$ is
    \[
        e_1 = \frac{f_1}{\norm{f_1}} = 1, \quad
        e_2 = \frac{f_2}{\norm{f_2}} = 2\sqrt{3}\left(x - \frac{1}{2}\right), \quad
        e_3 = \frac{f_3}{\norm{f_3}} = 6\sqrt{5}\left(x^2 - x + \frac{1}{6}\right).
    \]
    \item The differentiation operator $D$ on $\pcal_2(\rbb)$ has the following matrix with respect to the basis $1, x, x^2$:
    \[
        \mcal(D, (1, x, x^2)) =
        \begin{pmatrix}
            0 & 1 & 0 \\
            0 & 0 & 2 \\
            0 & 0 & 0
        \end{pmatrix}.
    \]
    Then
    \begin{align*}
        De_1 &= D(1) = 0, \\
        De_2 &= D\left(2\sqrt{3}\left(x - \frac{1}{2}\right)\right) = 2\sqrt{3} = 2\sqrt{3}e_1, \\
        De_3 &= D\left(6\sqrt{5}\left(x^2 - x + \frac{1}{6}\right)\right) = 6\sqrt{5}(2x - 1) = 6\sqrt{5} \cdot \frac{1}{\sqrt{3}}e_2 = 2\sqrt{15}e_2
    \end{align*}
    Hence, the matrix of $D$ with respect to the orthonormal basis $e_1, e_2, e_3$ is
    \[
        \mcal(D, (e_1, e_2, e_3)) =
        \begin{pmatrix}
            0 & 2\sqrt{3} & 0 \\
            0 & 0 & 2\sqrt{15} \\
            0 & 0 & 0
        \end{pmatrix},
    \]
    which is upper triangular, as expected.
\end{solenum}

\begin{exercise}{6b9}
    Suppose $e_1, \ldots, e_m$ is the result of applying the Gram--Schmidt procedure to a linearly independent list $v_1, \ldots, v_m$ in $V$.
    Prove that $\gen{v_k, e_k} > 0$ for each $k = 1, \ldots, m$.
\end{exercise}

\begin{sol}
    By the Gram--Schmidt procedure, we have
    \[
        e_k = \frac{f_k}{\norm{f_k}}, \quad f_k = v_k - \sum_{j=1}^{k-1}\gen{v_k, e_j}e_j.
    \]
    From the linear independence of $v_1, \ldots, v_m$, $v_k \notin \spn(v_1, \ldots, v_{k-1}) = \spn(e_1, \ldots, e_{k-1})$, so $f_k \neq 0$ and $\norm{f_k} > 0$.
    Then by orthogonality of $f_k$ and $e_j$ for $j < k$, we have
    \[
        \gen{v_k, e_k} = \frac{1}{\norm{f_k}}\gen{v_k, f_k} = \frac{1}{\norm{f_k}}\gen*{f_k + \sum_{j=1}^{k-1}\gen{v_k, e_j}e_j, f_k} = \frac{1}{\norm{f_k}}\gen{f_k, f_k} = \norm{f_k} > 0. \qedhere
    \]
\end{sol}

\begin{exercise}{6b10}
    Suppose $v_1, \ldots, v_m$ is a linearly independent list in $V$.
    Explain why the orthonormal list produced by the formulas of the Gram--Schmidt procedure (6.32) is the only orthonormal list $e_1, \ldots, e_m$ in $V$ such that $\gen{v_k, e_k} > 0$ and $\spn(v_1, \ldots, v_k) = \spn(e_1, \ldots, e_k)$ for each $k = 1, \ldots, m$.

    \note{The result in this exercise is used in the proof of 7.58.}
\end{exercise}

\begin{sol}
    Suppose $e_1, \ldots, e_m$ is an orthonormal list in $V$ such that $\gen{v_k, e_k} > 0$ and $\spn(v_1, \ldots, v_k) = \spn(e_1, \ldots, e_k)$ for each $k = 1, \ldots, m$.
    We induce on $k$ to show that $e_k$ agrees with the Gram--Schmidt output.

    For $k = 1$, we have $\spn(v_1) = \spn(e_1)$ and $\norm{e_1} = 1$, so $v_1 = \gen{v_1, e_1} e_1$.
    Taking norms and using $\gen{v_1, e_1} > 0$ gives $\gen{v_1, e_1} = \norm{v_1}$, so $e_1 = v_1 / \norm{v_1}$, which is the Gram--Schmidt output.

    Now suppose the claim holds for $e_1, \ldots, e_k$.
    Since $v_{k+1} \in \spn(e_1, \ldots, e_{k+1})$ and $e_1, \ldots, e_{k+1}$ is orthonormal,
    \[
        v_{k+1} = \sum_{j=1}^{k+1} \gen{v_{k+1}, e_j} e_j \implies \gen{v_{k+1}, e_{k+1}} e_{k+1} = v_{k+1} - \sum_{j=1}^{k} \gen{v_{k+1}, e_j} e_j.
    \]
    By the induction hypothesis, the right-hand side is the Gram--Schmidt vector $f_{k+1}$.
    Taking norms and using $\gen{v_{k+1}, e_{k+1}} > 0$ gives $\gen{v_{k+1}, e_{k+1}} = \norm{f_{k+1}}$, so $e_{k+1} = f_{k+1} / \norm{f_{k+1}}$, which is the Gram--Schmidt output.
\end{sol}

\begin{exercise}{6b11}
    Find a polynomial $q \in \pcal_2(\rbb)$ such that $p\left(\tfrac{1}{2}\right) = \int_0^1 pq$ for every $p \in \pcal_2(\rbb)$.
\end{exercise}

\begin{sol}
    The map $p \mapsto p(\frac{1}{2})$ is a linear functional on $\pcal_2(\rbb)$.
    By the Riesz representation theorem, there exists a unique $q \in \pcal_2(\rbb)$ such that $p(\frac{1}{2}) = \gen{p, q}$ for every $p \in \pcal_2(\rbb)$, given by
    \[
        q = e_1\!\left(\tfrac{1}{2}\right)e_1 + e_2\!\left(\tfrac{1}{2}\right)e_2 + e_3\!\left(\tfrac{1}{2}\right)e_3,
    \]
    where $e_1, e_2, e_3$ is the orthonormal basis from \exref{6b8}.
    Evaluating $e_1, e_2, e_3$ at $\frac{1}{2}$ gives
    \[
        e_1(\tfrac{1}{2}) = 1, \quad
        e_2(\tfrac{1}{2}) = 2\sqrt{3}\left(\frac{1}{2} - \frac{1}{2}\right) = 0, \quad
        e_3(\tfrac{1}{2}) = 6\sqrt{5}\left(\frac{1}{4} - \frac{1}{2} + \frac{1}{6}\right) = -\frac{\sqrt{5}}{2}.
    \]
    Hence, the polynomial $q$ is
    \[
        q = e_1 - \frac{\sqrt{5}}{2}e_3 = 1 - 15\left(x^2 - x + \frac{1}{6}\right) = -15x^2 + 15x - \frac{3}{2}. \qedhere
    \]
\end{sol}

\begin{exercise}{6b12}
    Find a polynomial $q \in \pcal_2(\rbb)$ such that
    \[
        \int_0^1 p(x) \cos(\pi x)\dd x = \int_0^1 pq
    \]
    for every $p \in \pcal_2(\rbb)$.
\end{exercise}

\begin{sol}
    By the Riesz representation theorem, there exists a unique $q \in \pcal_2(\rbb)$ such that
    \[
        \int_0^1 p(x) \cos(\pi x)\dd x = \gen{p, q} = \int_0^1 pq
    \]
    for every $p \in \pcal_2(\rbb)$, given by
    \[
        q = \gen{e_1, \cos(\pi x)}e_1 + \gen{e_2, \cos(\pi x)}e_2 + \gen{e_3, \cos(\pi x)}e_3,
    \]
    where $e_1, e_2, e_3$ is the orthonormal basis from \exref{6b8}.
    Evaluating the inner products gives
    \begin{align*}
        \gen{e_1, \cos(\pi x)} &= \int_0^1 e_1 \cos(\pi x) = \int_0^1 \cos(\pi x) = 0, \\
        \gen{e_2, \cos(\pi x)} &= \int_0^1 e_2 \cos(\pi x) = \int_0^1 2\sqrt{3}\left(x - \frac{1}{2}\right) \cos(\pi x) = \frac{-4\sqrt{3}}{\pi^2}, \\
        \gen{e_3, \cos(\pi x)} &= \int_0^1 e_3 \cos(\pi x) = \int_0^1 6\sqrt{5}\left(x^2 - x + \frac{1}{6}\right) \cos(\pi x) = 0.
    \end{align*}
    Hence,
    \[
        q = \frac{-4\sqrt{3}}{\pi^2} e_2 = \frac{-4\sqrt{3}}{\pi^2} \cdot 2\sqrt{3}\left(x - \frac{1}{2}\right) = \frac{-24}{\pi^2}\left(x - \frac{1}{2}\right). \qedhere
    \]
\end{sol}

\begin{exercise}{6b13}
    Show that a list $v_1, \ldots, v_m$ of vectors in $V$ is linearly dependent if and only if the Gram--Schmidt formula in 6.32 produces $f_k = 0$ for some $k \in \set{1, \ldots, m}$.

    \note{This exercise gives an alternative to Gaussian elimination techniques for determining whether a list of vectors in an inner product space is linearly dependent.}
\end{exercise}

\begin{sol}
    Let $f_k$, $e_k$ be the vectors obtained from the Gram--Schmidt procedure, with the process continuing as long as $f_k \neq 0$.
    If $v_1, \ldots, v_m$ is linearly dependent, take the smallest $k$ with $v_k \in \spn(v_1, \ldots, v_{k-1})$.
    Then $v_1, \ldots, v_{k-1}$ is linearly independent, so $e_1, \ldots, e_{k-1}$ is an orthonormal basis of that span.
    Hence,
    \[
        v_k = \sum_{j=1}^{k-1} \gen{v_k, e_j} e_j \implies f_k = v_k - \sum_{j=1}^{k-1} \gen{v_k, e_j} e_j = 0.
    \]

    Conversely, if the procedure gives $f_k = 0$ for some minimal $k$, then
    \[
        v_k = \sum_{j=1}^{k-1} \gen{v_k, e_j} e_j \in \spn(e_1, \ldots, e_{k-1}) = \spn(v_1, \ldots, v_{k-1}),
    \]
    so $v_1, \ldots, v_k$ is a linearly dependent list.
\end{sol}

\begin{exercise}{6b14}
    Suppose $V$ is a real inner product space and $v_1, \ldots, v_m$ is a linearly independent list of vectors in $V$.
    Prove that there exist exactly $2^m$ orthonormal lists $e_1, \ldots, e_m$ of vectors in $V$ such that
    \[
        \spn(v_1, \ldots, v_k) = \spn(e_1, \ldots, e_k)
    \]
    for all $k \in \set{1, \ldots, m}$.
\end{exercise}

\begin{sol}
    By \exref{6b10}, there is a unique orthonormal list $e_1, \ldots, e_m$ such that $\spn(v_1, \ldots, v_k) = \spn(e_1, \ldots, e_k)$ and $\gen{v_k, e_k} > 0$ for all $k$.
    Let $e_1', \ldots, e_m'$ be any orthonormal list satisfying the same span condition.
    For each $k$, the vector $e_k'$ lies in $\spn(e_1, \ldots, e_k)$ and is orthogonal to $\spn(e_1', \ldots, e_{k-1}') = \spn(e_1, \ldots, e_{k-1})$.
    Now, write $e_k'$ as
    \[
        e_k' = \sum_{j=1}^{k} c_j e_j
    \]
    for some scalars $c_1, \ldots, c_k$.
    Then $\gen{e_k', e_j} = 0$ for $j < k$ gives $c_j = 0$ for $j < k$, so $e_k' = c_k e_k$.
    Since $\norm{e_k'} = 1$ and $V$ is real, we have $e_k' = \pm e_k$.
    These $m$ sign choices are independent, giving exactly $2^m$ such lists.
\end{sol}

\begin{exercise}{6b15}
    Suppose $\gen{\cdot, \cdot}_1$ and $\gen{\cdot, \cdot}_2$ are inner products on $V$ such that $\gen{u, v}_1 = 0$ if and only if $\gen{u, v}_2 = 0$.
    Prove that there is a positive number $c$ such that $\gen{u, v}_1 = c\gen{u, v}_2$ for every $u, v \in V$.

    \note{This exercise shows that if two inner products have the same pairs of orthogonal vectors, then each of the inner products is a scalar multiple of the other inner product.}
\end{exercise}

\begin{sol}
    If $V = \set 0$, then the result holds trivially.
    Otherwise, assume $V \neq \set 0$, and let $e_1 \in V$ such that $\norm{e_1}_2 = 1$.
    Let $c = \norm{e_1}_1^2$.
    For any $u, v \in V$, then $U = \spn(u, v, e_1)$ is finite-dimensional.
    Extend $e_1$ to an orthonormal basis $e_1, \ldots, e_n$ of $U$ with respect to $\gen{\cdot, \cdot}_2$.
    Since $\gen{e_j, e_k}_2 = 0$ for $j \neq k$, the same holds for $\gen{\cdot, \cdot}_1$.

    For each $j > 1$, $\gen{e_1 + e_j, e_1 - e_j}_2 = 0$, so $\gen{e_1 + e_j, e_1 - e_j}_1 = 0$.
    Expanding gives
    \[
        0 = \gen{e_1 + e_j, e_1 - e_j}_1 = \gen{e_1, e_1}_1 - \gen{e_j, e_j}_1 = c - \gen{e_j, e_j}_1,
    \]
    so $\gen{e_j, e_j}_1 = c > 0$ for all $j$.
    Write $u = a_1 e_1 + \cdots + a_n e_n$ and $v = b_1 e_1 + \cdots + b_n e_n$ for some scalars $a_1, \ldots, a_n, b_1, \ldots, b_n$.
    Then
    \[
        \gen{u, v}_1 = \sum_{j=1}^{n} a_j\bar{b_j}\gen{e_j, e_j}_1 = c\sum_{j=1}^{n} a_j\bar{b_j} = c\gen{u, v}_2. \qedhere
    \]
\end{sol}

\begin{exercise}{6b16}
    Suppose $V$ is finite-dimensional.
    Suppose $\gen{\cdot, \cdot}_1$, $\gen{\cdot, \cdot}_2$ are inner products on $V$ with corresponding norms $\norm{\cdot}_1$ and $\norm{\cdot}_2$.
    Prove that there exists a positive number $c$ such that $\norm{v}_1 \leq c\norm{v}_2$ for every $v \in V$.
\end{exercise}

\begin{sol}
    If $V = \set 0$, then the result holds trivially.
    Otherwise, assume $V \neq \set 0$, and let $e_1, \ldots, e_n$ be an orthonormal basis of $V$ with respect to $\gen{\cdot, \cdot}_2$.
    For any $v = a_1 e_1 + \cdots + a_n e_n \in V$ for some scalars $a_1, \ldots, a_n$, we have
    \[
        \norm{v}_1 \leq \sum_{j=1}^{n} |a_j|\norm{e_j}_1 \leq \left(\sum_{j=1}^{n} |a_j|^2\right)^{1/2} \left(\sum_{j=1}^{n} \norm{e_j}_1^2\right)^{1/2} = \norm{v}_2 \left(\sum_{j=1}^{n} \norm{e_j}_1^2\right)^{1/2}.
    \]
    Taking $c = \left(\sum_{j=1}^{n} \norm{e_j}_1^2\right)^{1/2}$ gives the desired inequality.
\end{sol}

\begin{exercise}{6b17}
    Suppose $\fbb = \cbb$ and $V$ is finite-dimensional.
    Prove that if $T$ is an operator on $V$ such that $1$ is the only eigenvalue of $T$ and $\norm{Tv} \leq \norm{v}$ for all $v \in V$, then $T$ is the identity operator.
\end{exercise}

\begin{sol}
    By Schur's theorem, there exists an orthonormal basis $e_1, \ldots, e_n$ of $V$ such that $\mcal(T)$ is upper-triangular.
    Since $1$ is the only eigenvalue of $T$, the diagonal entries are all $1$, so
    \[
        Te_k = e_k + \sum_{i=1}^{k-1} a_{i,k} e_i
    \]
    for each $k$.
    By orthonormality,
    \[
        \norm{Te_k}^2 = \norm*{e_k + \sum_{i=1}^{k-1} a_{i,k} e_i}^2 = 1 + \sum_{i=1}^{k-1} |a_{i,k}|^2.
    \]
    Since $\norm{Te_k} \leq \norm{e_k} = 1$ by hypothesis, we have $\sum_{i=1}^{k-1} |a_{i,k}|^2 = 0$, forcing each $a_{i,k} = 0$.
    Thus, $Te_k = e_k$ for all $k$, so $T = I$.
\end{sol}

\begin{exercise}{6b18}
    Suppose $u_1, \ldots, u_m$ is a linearly independent list in $V$.
    Show that there exists $v \in V$ such that $\gen{u_k, v} = 1$ for all $k \in \set{1, \ldots, m}$.
\end{exercise}

\begin{sol}
    Let $U = \spn(u_1, \ldots, u_m)$.
    Since $u_1, \ldots, u_m$ is a basis of $U$, there exists a unique linear functional $f : U \to \fbb$ satisfying $f(u_k) = 1$ for all $k \in \set{1, \ldots, m}$.
    Because $U$ is a finite-dimensional inner product space (with the inner product inherited from $V$), the Riesz representation theorem provides a unique $v \in U$ such that $f(u) = \gen{u, v}$ for all $u \in U$.
    In particular, $\gen{u_k, v} = f(u_k) = 1$ for all $k$.
\end{sol}

\begin{exercise}{6b19}
    Suppose $v_1, \ldots, v_n$ is a basis of $V$.
    Prove that there exists a basis $u_1, \ldots, u_n$ of $V$ such that
    \[
        \gen{v_j, u_k} = 
        \begin{cases}
            0 & \text{if } j \neq k, \\
            1 & \text{if } j = k.
        \end{cases}
    \]
\end{exercise}

\begin{sol}
    For each $k \in \{1, \ldots, n\}$, since $v_1, \ldots, v_n$ is a basis of $V$, there is a unique linear functional $f_k : V \to \fbb$ satisfying $f_k(v_j) = \delta_{jk}$ for all $j \in \{1, \ldots, n\}$.
    By the Riesz representation theorem, there exists a unique $u_k \in V$ such that $f_k(v) = \gen{v, u_k}$ for all $v \in V$.
    In particular, $\gen{v_j, u_k} = f_k(v_j) = \delta_{jk}$.
    The list $u_1, \ldots, u_n$ is linearly independent because if $c_1 u_1 + \cdots + c_n u_n = 0$, then for each $j$,
    \[
        0 = \gen*{v_j, \sum_{k=1}^{n} c_k u_k} = \bar{c_j},
    \]
    which implies $c_j = 0$ for all $j$.
    Hence, $u_1, \ldots, u_n$ is a basis of $V$ with the desired properties.
\end{sol}

\begin{exercise}{6b20}
    Suppose $\fbb = \cbb$, $V$ is finite-dimensional, and $\ecal \subseteq \lin{V}$ is such that $ST = TS$ for all $S, T \in \ecal$.
    Prove that there is an orthonormal basis of $V$ with respect to which every element of $\ecal$ has an upper-triangular matrix.

    \note{This exercise strengthens \exref{5e9}(b) (in the context of inner product spaces) by asserting that the basis in that exercise can be chosen to be orthonormal.}
\end{exercise}

\begin{sol}
    By \exref{5e9}(b), there exists a basis $v_1, \ldots, v_n$ of $V$ such that every element of $\ecal$ has an upper-triangular matrix with respect to this basis.
    Equivalently, every $T \in \ecal$ satisfies $T(\spn(v_1, \ldots, v_k)) \subseteq \spn(v_1, \ldots, v_k)$ for each $k \in \{1, \ldots, n\}$.

    Apply the Gram--Schmidt procedure to $v_1, \ldots, v_n$ to obtain an orthonormal basis $e_1, \ldots, e_n$ of $V$.
    By the key property of Gram--Schmidt,
    \[
        \spn(e_1, \ldots, e_k) = \spn(v_1, \ldots, v_k)
    \]
    for each $k$.
    Hence, every $T \in \ecal$ still satisfies $T(\spn(e_1, \ldots, e_k)) \subseteq \spn(e_1, \ldots, e_k)$ for each $k$, which means every element of $\ecal$ has an upper-triangular matrix with respect to the orthonormal basis $e_1, \ldots, e_n$.
\end{sol}

\begin{exercise}{6b21}
    Suppose $\fbb = \cbb$, $V$ is finite-dimensional, $T \in \lin{V}$, and all eigenvalues of $T$ have absolute value less than $1$.
    Let $\ep > 0$.
    Prove that there exists a positive integer $m$ such that $\norm{T^m v} \leq \ep\norm{v}$ for every $v \in V$.

    \remark{This exercise is quite difficult with a long solution.
    A high level overview is this: Schur's theorem gives an orthonormal basis of $V$ with respect to which $\mcal(T)$ is upper-triangular.
    This reduces the problem to showing each basis vector is contracted by $T^m$ for sufficiently large $m$, which is done by induction on the index of the basis vector.
    The final step is to use the triangle inequality to show that $T^m$ contracts every vector in $V$.}
\end{exercise}

\begin{sol}
    If $V = \set 0$, then the result holds trivially.
    Otherwise, assume $V \neq \set 0$, and let $\dim V = n \geq 1$.
    By Schur's theorem, there exists an orthonormal basis $e_1, \ldots, e_n$ of $V$ such that $\mcal(T)$ is upper-triangular whose diagonal entries are the eigenvalues of $T$.
    Let $\lambda_1, \ldots, \lambda_n$ be the eigenvalues of $T$, so $|\lambda_j| < 1$ for each $j$.
    Write $Te_j = \lambda_j e_j + v_j$ for each $j$, where $v_j \in \spn(e_1, \ldots, e_{j-1})$ and $v_1 = 0$.
    Let
    \[
        \lambda = \max_j |\lambda_j| < 1,
    \]
    and choose $\alpha$ such that $\lambda < \alpha < 1$.

    We prove by induction on $j$ that there exists some $C_j > 0$ such that $\norm{T^k e_j} \leq C_j \alpha^k$ for all $k \geq 0$.
    For the base case $j = 1$, we have $Te_1 = \lambda_1 e_1$, which gives
    \[
        \norm{T^k e_1} = |\lambda_1|^k \leq \lambda^k < \alpha^k,
    \]
    so we can take $C_1 = 1$.

    Assume the inductive hypothesis holds for all indices less than $j$.
    Consider
    \[
        v_j = \sum_{i=1}^{j-1}a_i e_i.
    \]
    By the inductive hypothesis, $\norm{T^k e_i} \leq C_i \alpha^k$ for all $i < j$, so
    \[
        \norm{T^k v_j} \leq \sum_{i=1}^{j-1} |a_i|\,\norm{T^k e_i} \leq B\alpha^k, \quad B = \sum_{i=1}^{j-1} |a_i|C_i.
    \]
    Repeatedly applying $T$ where $Te_j = \lambda_j e_j + v_j$ gives
    \[
        T^k e_j = \lambda_j^k e_j + \sum_{i=0}^{k-1} \lambda_j^{k-1-i} T^i v_j.
    \]
    Taking norms and using the triangle inequality gives
    \[
        \norm{T^k e_j} \leq \lambda^k + B\sum_{i=0}^{k-1} \lambda^{k-1-i}\alpha^i = \lambda^k + B\frac{\alpha^k - \lambda^k}{\alpha - \lambda} < \left(1 + \frac{B}{\alpha - \lambda}\right)\alpha^k,
    \]
    so we can take $C_j = 1 + \frac{B}{\alpha - \lambda}$ to complete the inductive step.

    Let $C = \max\set{C_1, \ldots, C_n}$.
    For any $v \in V$, write $v = a_1 e_1 + \cdots + a_n e_n$ for some scalars $a_1, \ldots, a_n$.
    Then $|a_j| \leq \norm{v}$ by orthonormality, so
    \[
        \norm{T^k v} \leq \sum_{j=1}^{n} |a_j|\,\norm{T^k e_j} \leq nC\alpha^k\norm{v}.
    \]
    Choosing such $m$ that $nC\alpha^m \leq \ep$ gives $\norm{T^m v} \leq \ep\norm{v}$ for every $v \in V$.
\end{sol}

\begin{exercise}{6b22}
    Suppose $C[-1, 1]$ is the vector space of continuous real-valued functions on the interval $[-1, 1]$ with inner product given by
    \[
        \gen{f, g} = \int_{-1}^{1} fg
    \]
    for all $f, g \in C[-1, 1]$.
    Let $\phi$ be the linear functional on $C[-1, 1]$ defined by $\phi(f) = f(0)$.
    Show that there does not exist $g \in C[-1, 1]$ such that
    \[
        \phi(f) = \gen{f, g}
    \]
    for every $f \in C[-1, 1]$.

    \note{This exercise shows that the Riesz representation theorem (6.42) does not hold on infinite-dimensional vector spaces without additional hypotheses on $V$ and $\phi$.}
\end{exercise}

\begin{sol}
    Suppose for contradiction that there exists $g \in C[-1, 1]$ such that $f(0) = \int_{-1}^{1} fg$ for every $f \in C[-1, 1]$.
    Take $f(x) = x^2 g(x)$, which is continuous.
    Then
    \[
        0 = f(0) = \int_{-1}^{1} x^2 (g(x))^2 \dd x.
    \]
    The integrand is continuous and nonnegative, so $x^2 (g(x))^2 = 0$ for all $x \in [-1, 1]$.
    Hence, $g(x) = 0$ for all $x \in [-1, 1] \setminus \set{0}$, and continuity forces $g(0) = 0$, so $g \equiv 0$.
    But then $f(0) = 0$ for every $f \in C[-1, 1]$, which is false (take $f = 1$), a contradiction.
    Hence, no such $g$ exists.
\end{sol}

\begin{exercise}{6b23}
    For all $u, v \in V$, define $d(u, v) = \norm{u - v}$.
    \begin{subexercises}
        \item Show that $d$ is a metric on $V$.
        \item Show that if $V$ is finite-dimensional, then $d$ is a complete metric on $V$ (meaning that every Cauchy sequence converges).
        \item Show that every finite-dimensional subspace of $V$ is a closed subset of $V$ (with respect to the metric $d$).
    \end{subexercises}
    \note{This exercise requires familiarity with metric spaces.}
\end{exercise}

\begin{solenum}
    \item We verify the metric axioms for $d$:
    \begin{itemize}
        \item \textbf{Non-negativity and Identity}: For all $u, v \in V$, $\norm{u - v} \geq 0$ and $\norm{u - v} = 0$ if and only if $u = v$.
        \item \textbf{Symmetry}: For all $u, v \in V$, we have
        \[
            d(u, v) = \norm{u - v} = \norm{-(v - u)} = \norm{v - u} = d(v, u).
        \]
        \item \textbf{Triangle Inequality}: For all $u, v, w \in V$, we have
        \[
            d(u, w) = \norm{u - w} = \norm{(u - v) + (v - w)} \leq \norm{u - v} + \norm{v - w} = d(u, v) + d(v, w).
        \]
    \end{itemize}
    Therefore, $(V, d)$ is a metric space.
    \item If $V$ is finite-dimensional, let $e_1, \ldots, e_n$ be an orthonormal basis of $V$.
    A Cauchy sequence $(v_k)_{k \geq 1}$ in $V$ can be written as $v_k = a_{k,1} e_1 + \cdots + a_{k,n} e_n$ for scalars $a_{k,j}$.
    By the Cauchy--Schwarz inequality, for each $j$,
    \[
        |a_{k,j} - a_{m,j}| = |\gen{v_k - v_m, e_j}| \leq \norm{v_k - v_m},
    \]
    so each coordinate sequence $(a_{k,j})_{k \geq 1}$ is Cauchy in $\fbb$.
    Since $\fbb$ is complete, each $(a_{k,j})$ converges to some $a_j \in \fbb$.
    Define $v = a_1 e_1 + \cdots + a_n e_n \in V$.
    Since $e_1, \ldots, e_n$ is orthonormal,
    \[
        \norm{v_k - v}^2 = \sum_{j=1}^{n} |a_{k,j} - a_j|^2 \to 0,
    \]
    so $v_k \to v$ in $V$, and $(V, d)$ is complete.
    \item Let $U$ be a finite-dimensional subspace of $V$, and let $(u_k)_{k \geq 1}$ be a sequence in $U$ converging to some $v \in V$.
    Since $(u_k)$ converges, it is Cauchy.
    Restricting the inner product on $V$ to $U$, we see that $(u_k)$ is also a Cauchy sequence in $U$.
    By part (b), $U$ is complete, so $(u_k)$ converges to some $u \in U$.
    Since $(u_k)$ converges to $v$ in $V$ and to $u$ in $U$, we have $v = u \in U$.
    Therefore, $U$ is closed in $V$.
\end{solenum}
