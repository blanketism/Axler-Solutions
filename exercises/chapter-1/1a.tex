\section{Vector Spaces}

\subsection{\texorpdfstring{$\rbb^n$ and $\cbb^n$}{Rn and Cn}}

\begin{exercise}{1a1}
    Show that $\alpha + \beta = \beta + \alpha$ for all $\alpha, \beta \in \cbb$.
\end{exercise}

\begin{sol}
    Let $\alpha = a + bi$ and $\beta = c + di$.
    Then
    \begin{align*}
        \alpha + \beta & = (a + bi) + (c + di) \\
        & = (a + c) + (b + d)i \\
        & = (c + a) + (d + b)i \\
        & = (c + di) + (a + bi) = \beta + \alpha. \qedhere
    \end{align*}
\end{sol}

\begin{exercise}{1a2}
    Show that $(\alpha + \beta) + \gamma = \alpha + (\beta + \gamma)$ for all $\alpha, \beta, \gamma \in \cbb$.
\end{exercise}

\begin{sol}
    Let $\alpha = a + bi$, $\beta = c + di$, and $\gamma = e + fi$.
	Then
    \begin{align*}
        (\alpha + \beta) + \gamma & = ((a + bi) + (c + di)) + (e + fi) \\
        & = ((a + c) + (b + d)i) + (e + fi) \\
        & = (a + c + e) + (b + d + f)i \\
        & = (a + (c + e)) + (b + (d + f))i \\
        & = (a + bi) + ((c + di) + (e + fi)) = \alpha + (\beta + \gamma). \qedhere
    \end{align*}
\end{sol}

\begin{exercise}{1a3}
    Show that $(\alpha\beta)\gamma = \alpha(\beta\gamma)$ for all $\alpha, \beta, \gamma \in \cbb$.
\end{exercise}

\begin{sol}
    Let $\alpha = a + bi$, $\beta = c + di$, and $\gamma = e + fi$.
	Then
    \begin{align*}
        (\alpha\beta)\gamma & = ((a + bi)(c + di))(e + fi) \\
        & = ((ac - bd) + (ad + bc)i)(e + fi) \\
        & = (ace - bde - adf - bcf) + (acf - bdf + ade + bce)i \\
        & = (ace - adf - bde - bcf) + (acf + ade + bce - bdf)i \\
        & = (a + bi)((ce - df) + (cf + de)i) \\
        & = (a + bi)((c + di)(e + fi)) = \alpha(\beta\gamma). \qedhere
    \end{align*}
\end{sol}

\begin{exercise}{1a4}
    Show that $\gamma(\alpha + \beta) = \gamma\alpha + \gamma\beta$ for all $\alpha, \beta, \gamma \in \cbb$.
\end{exercise}

\begin{sol}
    Let $\alpha = a + bi$, $\beta = c + di$, and $\gamma = e + fi$.
	Then
    \begin{align*}
        \gamma(\alpha + \beta) & = (e + fi)((a + bi) + (c + di)) \\
        & = (e + fi)((a + c) + (b + d)i) \\
        & = (e(a + c) - f(b + d)) + (e(b + d) + f(a + c))i \\
        & = (ea - fb + ec - fd) + (eb + fa + ed + fc)i \\
        & = (ea - fb) + (eb + fa)i + (ec - fd) + (ed + fc)i \\
        & = (e + fi)(a + bi) + (e + fi)(c + di) = \gamma\alpha + \gamma\beta. \qedhere
    \end{align*}
\end{sol}

\begin{exercise}{1a5}
    Show that for every $\alpha \in \cbb$, there exists a unique $\beta \in \cbb$ such that $\alpha + \beta = 0$.
\end{exercise}

\begin{sol}
    Let $\alpha = a + bi$.
    Then $\beta = -a - bi$ satisfies
    \[
        \alpha + \beta = (a + bi) + (-a - bi) = (a - a) + (b - b)i = 0 + 0i = 0.
    \]
    If $\gamma \in \cbb$ is such that $\alpha + \gamma = 0$, then \exref{1a1} and \exref{1a2} give
    \[
        \gamma = \gamma + 0 = \gamma + (\alpha + \beta) = (\gamma + \alpha) + \beta = 0 + \beta = \beta.
    \]
    Therefore, $\beta$ is unique.
\end{sol}

\begin{exercise}{1a6}
    Show that for every $\alpha \in \cbb$ with $\alpha \neq 0$, there exists a unique $\beta \in \cbb$ such that $\alpha\beta = 1$.
\end{exercise}

\begin{sol}
    Let $\alpha = a + bi$ be such that $\alpha \neq 0$, so at least one of $a$ or $b$ is nonzero.
    Then $a^2 + b^2 \neq 0$.
    Let $\beta \in \cbb$ be given by
    \[
        \beta = \frac{a}{a^2 + b^2} + \frac{-b}{a^2 + b^2}i.
    \]
    Since $(a + bi)(a - bi) = a^2 + b^2$, we have
    \[
        \alpha\beta = (a + bi)\left(\frac{a}{a^2 + b^2} + \frac{-b}{a^2 + b^2}i\right) = \frac{(a + bi)(a - bi)}{a^2 + b^2} = \frac{a^2 + b^2}{a^2 + b^2} = 1.
    \]
    Moreover, if $\gamma \in \cbb$ is such that $\alpha\gamma = 1$, then using \exref{1a3} gives
    \[
        \gamma = \gamma \cdot 1 = \gamma(\alpha\beta) = (\gamma\alpha)\beta = 1 \cdot \beta = \beta.
    \]
    Therefore, $\beta$ is unique.
\end{sol}

\begin{exercise}{1a7}
    Show that
    \[
    	\frac{-1 + \sqrt3i}{2}
    \]
    is a cube root of $1$ (meaning that its cube equals $1$).
\end{exercise}

\begin{sol}
    By direct computation,
    \[
        \left(\frac{-1 + \sqrt3i}{2}\right)^3 = \frac{(-1)^3 + 3(-1)^2(\sqrt3i) + 3(-1)(\sqrt3i)^2 + (\sqrt3i)^3}{8} = \frac{-1 + 3\sqrt{3}i + 9 - 3\sqrt{3}i}{8} = \frac{8}{8} = 1. \qedhere
    \]
\end{sol}

\begin{exercise}{1a8}
    Find two distinct square roots of $i$.
\end{exercise}

\begin{sol}
    Let $z = \frac{1+i}{\sqrt 2}$.
    Then
    \[
        z^2 = \frac{1+2i-1}{2} = \frac{2i}{2} = i.
    \]
    Since $z \neq 0$, it follows that $-z \neq z$, and $(-z)^2 = (-1)^2 z^2 = z^2 = i$, so $-z$ is also a square root of $i$.
\end{sol}

\begin{exercise}{1a9}
    Find $x \in \rbb^4$ such that
    \[
    	(4, -3, 1, 7) + 2x = (5, 9, -6, 8).
    \]
\end{exercise}

\begin{sol}
    By direct computation,
    \[
        x = \frac{1}{2}((5, 9, -6, 8) - (4, -3, 1, 7)) = \left(\frac{1}{2}, 6, -\frac{7}{2}, \frac{1}{2}\right). \qedhere
    \]
\end{sol}

\begin{exercise}{1a10}
    Explain why there does not exist $\lambda \in \cbb$ such that
    \[
    	\lambda(2 - 3i, 5 + 4i, -6 + 7i) = (12 - 5i, 7 + 22i, -32 - 9i).
    \]
\end{exercise}

\begin{sol}
    If such a $\lambda$ exists, then $\lambda(2 - 3i) = 12 - 5i$ gives
    \[
        \lambda = \frac{12 - 5i}{2 - 3i} = 3 + 2i.
    \]
    However, $\lambda(-6 + 7i) = (3 + 2i)(-6 + 7i) = -32 + 9i \neq -32 - 9i$, so no such $\lambda$ exists.
\end{sol}

\begin{exercise}{1a11}
    Show that $(x + y) + z = x + (y + z)$ for all $x, y, z \in \fbb^n$.
\end{exercise}

\begin{sol}
    Let $x = (x_1, \dots, x_n), y = (y_1, \dots, y_n), z = (z_1, \dots, z_n) \in \fbb^n$.
    Then
    \begin{align*}
        (x + y) + z &= (x_1 + y_1, \dots, x_n + y_n) + (z_1, \dots, z_n) \\
        &= ((x_1 + y_1) + z_1, \dots, (x_n + y_n) + z_n) \\
        &= (x_1 + (y_1 + z_1), \dots, x_n + (y_n + z_n)) \\
        &= (x_1, \dots, x_n) + (y_1 + z_1, \dots, y_n + z_n) = x + (y + z). \qedhere
    \end{align*}
\end{sol}

\begin{exercise}{1a12}
    Show that $(ab)x = a(bx)$ for all $a, b \in \fbb$ and $x \in \fbb^n$.
\end{exercise}

\begin{sol}
    Let $a, b \in \fbb$ and let $x = (x_1, \dots, x_n) \in \fbb^n$.
    Then
    \begin{align*}
        (ab)x &= (ab)(x_1, \dots, x_n) \\
        &= ((ab)x_1, \dots, (ab)x_n) \\
        &= (a(bx_1), \dots, a(bx_n)) \\
        &= a(bx_1, \dots, bx_n) \\
        &= a(b(x_1, \dots, x_n)) = a(bx). \qedhere
    \end{align*}
\end{sol}

\begin{exercise}{1a13}
    Show that $1x = x$ for all $x \in \fbb^n$.
\end{exercise}

\begin{sol}
    Let $x = (x_1, \dots, x_n) \in \fbb^n$.
    Then
    \begin{align*}
        1x &= 1(x_1, \dots, x_n) \\
        &= (1x_1, \dots, 1x_n) \\
        &= (x_1, \dots, x_n) = x. \qedhere
    \end{align*}
\end{sol}

\begin{exercise}{1a14}
    Show that $\lambda(x + y) = \lambda x + \lambda y$ for all $\lambda \in \fbb$ and $x, y \in \fbb^n$.
\end{exercise}

\begin{sol}
    Let $\lambda \in \fbb$ and let $x = (x_1, \dots, x_n), y = (y_1, \dots, y_n) \in \fbb^n$.
    Then
    \begin{align*}
        \lambda(x + y) &= \lambda(x_1 + y_1, \dots, x_n + y_n) \\
        &= (\lambda(x_1 + y_1), \dots, \lambda(x_n + y_n)) \\
        &= (\lambda x_1 + \lambda y_1, \dots, \lambda x_n + \lambda y_n) \\
        &= (\lambda x_1, \dots, \lambda x_n) + (\lambda y_1, \dots, \lambda y_n) \\
        &= \lambda(x_1, \dots, x_n) + \lambda(y_1, \dots, y_n) = \lambda x + \lambda y. \qedhere
    \end{align*}
\end{sol}

\begin{exercise}{1a15}
    Show that $(a + b)x = ax + bx$ for all $a, b \in \fbb$ and $x \in \fbb^n$.
\end{exercise}

\begin{sol}
    Let $a, b \in \fbb$ and let $x = (x_1, \dots, x_n) \in \fbb^n$.
    Then
    \begin{align*}
        (a + b)x &= (a + b)(x_1, \dots, x_n) \\
        &= ((a + b)x_1, \dots, (a + b)x_n) \\
        &= (ax_1 + bx_1, \dots, ax_n + bx_n) \\
        &= (ax_1, \dots, ax_n) + (bx_1, \dots, bx_n) \\
        &= a(x_1, \dots, x_n) + b(x_1, \dots, x_n) = ax + bx. \qedhere
    \end{align*}
\end{sol}
