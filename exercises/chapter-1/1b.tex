\subsection{Definition of Vector Space}

\begin{exercise}{1b1}
    Prove that $-(-v) = v$ for every $v \in V$.
\end{exercise}

\begin{sol}
    Let $v \in V$.
	By the definition of additive inverse, we have
    \[
    	v + (-v) = 0 = -(-v) + (-v),
    \]
    so both $v$ and $-(-v)$ are additive inverses of $-v$.
    By the uniqueness of additive inverses, we conclude that $-(-v) = v$.
\end{sol}

\begin{exercise}{1b2}
    Suppose $a \in \fbb, v \in V$, and $av = 0$.
	Prove that $a = 0$ or $v = 0$.
\end{exercise}

\begin{sol}
    If $a = 0$, then we are done.
	Suppose $a \neq 0$.
	Then
    \[
    	v = 1v = \frac1a(av) = \frac1a \cdot 0 = 0. \qedhere
    \]
\end{sol}

\begin{exercise}{1b3}
    Suppose $v, w \in V$.
	Explain why there exists a unique $x \in V$ such that $v + 3x = w$.
\end{exercise}

\begin{sol}
    Let $x = \frac{1}{3}(w - v)$.
    Then
    \[
        v + 3\left(\frac{1}{3}(w - v)\right) = v + (w - v) = w.
    \]
    If $y \in V$ is such that $v + 3y = w$, then
    \[
        v + 3y = w = v + 3x \implies 3y = 3x \implies y = x.
    \]
    Therefore, $x$ is the unique element of $V$ such that $v + 3x = w$.
\end{sol}

\begin{exercise}{1b4}
    The empty set is not a vector space.
	The empty set fails to satisfy only one of the requirements listed in the definition of a vector space.
	Which one?
\end{exercise}

\begin{sol}
    The empty set fails to satisfy the requirement that there exists a zero vector $0 \in V$ such that $v + 0 = v$ for all $v \in V$ since it contains no elements.
\end{sol}

\begin{exercise}{1b5}
    Show that in the definition of a vector space, the additive inverse condition can be replaced with the condition that
    \[
    	0v = 0 \text{ for all } v \in V.
    \]
    Here, the $0$ on the left side is the number $0$, and the $0$ on the right side is the additive identity in $V$.

    \note{The phrase a ``condition can be replaced'' in a definition means that the collection of objects satisfying the definition is unchanged if the original condition is replaced with the new condition.}
\end{exercise}

\begin{sol}
    Suppose that $V$ satisfies all the vector space axioms except possibly the additive inverse condition.
    We must prove that the additive inverse condition holds if and only if $0v = 0$ for all $v \in V$.

    \rightimp Suppose the additive inverse condition holds.
    By 1.30, $0v = 0$ for all $v \in V$.

    \leftimp Suppose $0v = 0$ for all $v \in V$.
    Let $v \in V$.
    Then
    \[
        v + (-1)v = 1v + (-1)v = (1 + (-1))v = 0v = 0.
    \]
    Hence, $(-1)v$ is the additive inverse of $v$.
\end{sol}

\begin{exercise}{1b6}
    Let $\infty$ and $-\infty$ denote two distinct objects, neither of which is in $\rbb$.
	Define an addition and scalar multiplication on $\rbb \cup \set{\infty, -\infty}$ as you could guess from the notation.
	Specifically, the sum and product of two real numbers is as usual, and for $t \in \rbb$, define
    \[
        t\infty = 
        \begin{cases}
            -\infty & \text{if } t < 0, \\
            0 & \text{if } t = 0, \\
            \infty & \text{if } t > 0,
        \end{cases} \quad
        t(-\infty) = 
        \begin{cases}
            \infty & \text{if } t < 0, \\
            0 & \text{if } t = 0, \\
            -\infty & \text{if } t > 0,
        \end{cases}
    \]
    and
    \begin{align*}
        t + \infty & = \infty + t = \infty + \infty = \infty, \\
        t + (-\infty) & = (-\infty) + t = (-\infty) + (-\infty) = -\infty, \\
        \infty + (-\infty) & = (-\infty) + \infty = 0.
    \end{align*}
    With these operations of addition and scalar multiplication, is $\rbb \cup \set{\infty, -\infty}$ a vector space over $\rbb$?
	Explain.
\end{exercise}

\begin{sol}
    No, $\rbb \cup \set{\infty, -\infty}$ is not a vector space over $\rbb$ as it does not satisfy the associative property of addition.
	Let $t \in \rbb \setminus \set 0$.
	Then
    \[
    	(t + \infty) + (-\infty) = \infty + (-\infty) = 0 \neq t = t + 0 = t + (\infty + (-\infty)). \qedhere
    \]
\end{sol}

\begin{exercise}{1b7}
    Suppose $S$ is a nonempty set.
	Let $V^S$ denote the set of functions from $S$ to $V$.
	Define a natural addition and scalar multiplication on $V^S$, and show that $V^S$ is a vector space with these definitions.
\end{exercise}

\begin{sol}
    Let $f, g \in V^S$ and let $a \in \fbb$.
    Define addition and scalar multiplication on $V^S$ by
    \[
        (f + g)(x) = f(x) + g(x), \quad (af)(x) = a(f(x)),
    \]
    for all $x \in S$.
    We now verify the vector space axioms for $V^S$, where the following calculations repeatedly use the commutativity, associativity, and distributive properties of addition and scalar multiplication in $V$ and $\fbb$.
    \begin{enumerate}
        \item \textbf{Commutativity.} Let $f, g \in V^S$.
        Then for all $x \in S$,
        \[
            (f + g)(x) = f(x) + g(x) = g(x) + f(x) = (g + f)(x),
        \]
        so $f + g = g + f$.
        \item \textbf{Associativity.} Let $f, g, h \in V^S$ and let $a, b \in \fbb$.
        Then for all $x \in S$,
        \[
            ((f + g) + h)(x) = (f + g)(x) + h(x) = (f(x) + g(x)) + h(x) = f(x) + (g(x) + h(x)) = (f + (g + h))(x),
        \]
        so $(f + g) + h = f + (g + h)$.
        Moreover,
        \[
            ((ab)f)(x) = (ab)f(x) = a(bf(x)) = a((bf)(x)) = (a(bf))(x),
        \]
        so $(ab)f = a(bf)$.
        \item \textbf{Additive identity.} Let $0 \in V$ be the additive identity in $V$, and define the zero function $0 \in V^S$ by $0(x) = 0$ for all $x \in S$.
        Then for all $f \in V^S$ and all $x \in S$,
        \[
            (f + 0)(x) = f(x) + 0 = f(x),
        \]
        so $f + 0 = f$.
        \item \textbf{Additive inverse.} Let $f \in V^S$.
        Define $-f \in V^S$ by $(-f)(x) = -f(x)$ for all $x \in S$.
        Then for all $x \in S$,
        \[
            (f + (-f))(x) = f(x) + (-f(x)) = 0 = 0(x),
        \]
        so $f + (-f) = 0$.
        \item \textbf{Multiplicative identity.} Let $1 \in \fbb$ be the multiplicative identity in $\fbb$.
        Then for all $f \in V^S$ and all $x \in S$,
        \[
            (1f)(x) = 1 \cdot f(x) = f(x),
        \]
        so $1f = f$.
        \item \textbf{Distributive properties.} Let $a, b \in \fbb$ and $f, g \in V^S$.
        Then for all $x \in S$,
        \[
            (a(f + g))(x) = a((f + g)(x)) = a(f(x) + g(x)) = af(x) + ag(x) = (af + ag)(x),
        \]
        so $a(f + g) = af + ag$.

        Also,
        \[
            ((a + b)f)(x) = (a + b)f(x) = af(x) + bf(x) = (af + bf)(x),
        \]
        so $(a + b)f = af + bf$.
    \end{enumerate}
    Therefore, $V^S$ is a vector space.
\end{sol}

\begin{exercise}{1b8}
    Suppose $V$ is a real vector space.
    \begin{itemize}
        \item The \term{complexification} of $V$, denoted by $V_{\cbb}$, equals $V \times V$.
	    An element of $V_{\cbb}$ is an ordered pair $(u, v)$, where $u, v \in V$, but we write this as $u + iv$.
        \item Addition on $V_{\cbb}$ is defined by
        \[
        	(u_1 + iv_1) + (u_2 + iv_2) = (u_1 + u_2) + i(v_1 + v_2)
        \]
        for all $u_1, u_2, v_1, v_2 \in V$.
        \item Complex scalar multiplication on $V_{\cbb}$ is defined by
        \[
        	(a + ib)(u + iv) = (au - bv) + i(av + bu)
        \]
        for all $a, b \in \rbb$ and $u, v \in V$.
    \end{itemize}
    Prove that with the definitions of addition and scalar multiplication as above, $V_{\cbb}$ is a complex vector space.

    \note{Think of $V$ as a subset of $V_{\cbb}$ by identifying $u \in V$ with $u +i0$.
    The construction of $V_{\cbb}$ from $V$ can then be thought of as generalizing the construction of $\cbb^n$ from $\rbb^n$.}
\end{exercise}

\begin{sol}
    We prove that $V_{\cbb}$ is a complex vector space by verifying the vector space axioms.
    \begin{enumerate}
        \item \textbf{Commutativity.} Let $u_1 + iv_1, u_2 + iv_2 \in V_{\cbb}$.
        Then
        \[
            (u_1 + iv_1) + (u_2 + iv_2) = (u_1 + u_2) + i(v_1 + v_2) = (u_2 + u_1) + i(v_2 + v_1) = (u_2 + iv_2) + (u_1 + iv_1).
        \]
        \item \textbf{Associativity.} Let $u_1 + iv_1, u_2 + iv_2, u_3 + iv_3 \in V_{\cbb}$ and let $a + ib, c + id \in \cbb$.
        Then
        \begin{align*}
            ((u_1 + iv_1) + (u_2 + iv_2)) + (u_3 + iv_3) &= ((u_1 + u_2) + i(v_1 + v_2)) + (u_3 + iv_3) \\
            &= ((u_1 + u_2) + u_3) + i((v_1 + v_2) + v_3) \\
            &= (u_1 + (u_2 + u_3)) + i(v_1 + (v_2 + v_3)) \\
            &= (u_1 + iv_1) + ((u_2 + u_3) + i(v_2 + v_3)) \\
            &= (u_1 + iv_1) + ((u_2 + iv_2) + (u_3 + iv_3)).
        \end{align*}
        Moreover,
        \begin{align*}
            ((a + ib)(c + id))(u_1 + iv_1) &= ((ac - bd) + i(ad + bc))(u_1 + iv_1) \\
            &= ((ac - bd)u_1 - (ad + bc)v_1) + i((ad + bc)u_1 + (ac - bd)v_1) \\
            &= (a(cu_1 - dv_1) - b(cv_1 + du_1)) + i(a(cv_1 + du_1) + b(cu_1 - dv_1)) \\
            &= (a + ib)((c + id)(u_1 + iv_1)).
        \end{align*}
        \item \textbf{Additive identity.} Let $0 \in V$ be the additive identity in $V$.
        Then for any $u + iv \in V_{\cbb}$,
        \[
            (0 + i0) + (u + iv) = (0 + u) + i(0 + v) = u + iv,
        \]
        so $0 + i0$ is the additive identity in $V_{\cbb}$.
        \item \textbf{Additive inverse.} Let $u + iv \in V_{\cbb}$.
        Since $V$ is a vector space, both $u$ and $v$ have additive inverses $-u, -v \in V$, respectively.
        Then
        \[
            (u + iv) + ((-u) + i(-v)) = (u + (-u)) + i(v + (-v)) = 0 + i0,
        \]
        so $(-u) + i(-v)$ is the additive inverse of $u + iv$ in $V_{\cbb}$.
        \item \textbf{Multiplicative identity.} Write $1 = 1 + i0 \in \cbb$.
        Then for any $u + iv \in V_{\cbb}$,
        \[
            (1 + i0)(u + iv) = (1 \cdot u - 0 \cdot v) + i(1 \cdot v + 0 \cdot u) = u + iv.
        \]
        \item \textbf{Distributive properties.} Let $u_1 + iv_1, u_2 + iv_2 \in V_{\cbb}$ and let $a + ib, c + id \in \cbb$.
        Then
        \begin{align*}
            (a + ib)((u_1 + iv_1) + (u_2 + iv_2)) &= (a + ib)((u_1 + u_2) + i(v_1 + v_2)) \\
            &= (a(u_1 + u_2) - b(v_1 + v_2)) + i(a(v_1 + v_2) + b(u_1 + u_2)) \\
            &= (au_1 - bv_1 + au_2 - bv_2) + i(av_1 + bu_1 + av_2 + bu_2) \\
            &= (au_1 - bv_1 + i(av_1 + bu_1)) + (au_2 - bv_2 + i(av_2 + bu_2)) \\
            &= (a + ib)(u_1 + iv_1) + (a + ib)(u_2 + iv_2).
        \end{align*}
        Moreover,
        \begin{align*}
            ((a + ib) + (c + id))(u_1 + iv_1) &= ((a + c) + i(b + d))(u_1 + iv_1) \\
            &= ((a + c)u_1 - (b + d)v_1) + i((a + c)v_1 + (b + d)u_1) \\
            &= (au_1 - bv_1 + cu_1 - dv_1) + i(av_1 + bu_1 + cv_1 + du_1) \\
            &= (au_1 - bv_1 + i(av_1 + bu_1)) + (cu_1 - dv_1 + i(cv_1 + du_1)) \\
            &= (a + ib)(u_1 + iv_1) + (c + id)(u_1 + iv_1).
        \end{align*}
    \end{enumerate}
    Therefore, $V_{\cbb}$ is a complex vector space.
\end{sol}
